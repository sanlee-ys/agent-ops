## Code Review Rules

An automated GitHub review reads this section. Follow it when you review a
pull request in this fleet.

Review a diff only for:

- **A bug** — code that does not do what it claims, or that fails on a
  reachable input.
- **A regression** — a behavior change that breaks an existing guarantee.
- **A requirement mismatch** — the diff does not do what the linked issue or
  the PR description asks for.
- **A missing or weakened test** — a change that ships with no test, or a
  test that no longer proves the behavior it claims to cover.
- **A scope-fidelity question** — state plainly whether this diff adds
  anything unrequested, or drops anything the request asked for.

     1 +#!/usr/bin/env python3
     2 +"""Ground-truth snapshot of a repo, for reconciling an agent's claims against it.
     3 +
     4 +THE PROBLEM. An agent's self-report is a CLAIM, not a record. "Opened PR #51",
     5 +"merged it", "pushed the branch" are sentences the agent wrote, and the agent
     6 +writes the same sentence whether the `gh` call succeeded, returned an error it
     7 +did not read, or never ran. `conventions/agent-success-signals.md` records the
     8 +same shape from the tool side: a green check means the step exited 0, not that
     9 +the work happened.
    10 +
    11 +THE ANSWER IS MECHANICAL, NOT RHETORICAL. A session or an agentic loop compares
    12 +its own claims against this snapshot at the end of a cycle. The snapshot comes
    13 +from the systems of record - `gh` for the forge, `git` for the tree and the
    14 +remote - and never from the transcript. A claim with no matching record in the
    15 +snapshot is a fabrication or a silent failure, and the two are the same problem
    16 +for the reader: the report says work exists that does not.
    17 +
    18 +WHAT IT COLLECTS, per repo:
    19 +  - open pull requests: number, title, head branch, creation time
    20 +  - pull requests merged since the window start
    21 +  - branches that ACTUALLY exist on the remote, from `git ls-remote`
    22 +  - the current branch
    23 +  - uncommitted paths, from `git status --porcelain`
    24 +  - the last local commit
    25 +
    26 +`git ls-remote` rather than `git branch -r` is deliberate and is the one place
    27 +this script would be wrong if it followed the obvious route. `git branch -r`
    28 +prints a LOCAL cache of remote-tracking refs. It goes stale the moment another
    29 +machine or another session moves the remote, and it happily lists a branch that
    30 +was deleted an hour ago. A snapshot built to catch a false claim must not be
    31 +built from a cache that can carry one.
    32 +
    33 +OUTPUT. JSON on stdout, a compact human table on stderr. The split is the
    34 +interface: pipe stdout into a comparison and read stderr yourself.
    35 +
    36 +EXIT CODES are the interface: 0 snapshot complete, 1 one or more repos failed
    37 +(the JSON still carries the ones that worked, each failure named in its entry),
    38 +2 usage error.
    39 +
    40 +It READS ONLY. It runs no command that writes, and it sends nothing anywhere.
    41 +"""
    42 +from __future__ import annotations
    43 +
    44 +import argparse
    45 +import json
    46 +import re
    47 +import subprocess
    48 +import sys
    49 +from datetime import datetime, timedelta, timezone
    50 +
    51 +# A duration like `6h`, `90m`, `2d`, `3w`. Anything else is parsed as ISO-8601.
    52 +_DURATION = re.compile(r"^(\d+)\s*([smhdw])$", re.IGNORECASE)
    53 +_UNIT_SECONDS = {"s": 1, "m": 60, "h": 3600, "d": 86400, "w": 604800}
    54 +
    55 +# `git ls-remote --heads` prints "<sha>\trefs/heads/<branch>".
    56 +_LS_REMOTE_LINE = re.compile(r"^([0-9a-f]{7,40})\s+refs/heads/(.+)$")
    57 +
    58 +# `git status --porcelain` prints "XY <path>", and a rename prints
    59 +# "R  <old> -> <new>". The path starts at column 3.
    60 +_RENAME_ARROW = " -> "
    61 +
    62 +_TIMEOUT = 30
    63 +
    64 +
    65 +class RepoError(RuntimeError):
    66 +    """A repo could not be snapshotted. Named, never swallowed."""

    69 +# --- Window -----------------------------------------------------------------

    72 +def parse_since(value: str, now: datetime | None = None) -> datetime:
    73 +    """An ISO-8601 datetime, or a duration before now (`6h`, `2d`, `90m`).
    74 +
    75 +    A naive ISO value is read as UTC. The caller supplies `now` in tests so the
    76 +    parsing is deterministic without freezing the clock.
    77 +    """
    78 +    now = now or datetime.now(timezone.utc)
    79 +    text = value.strip()
    80 +    if not text:
    81 +        raise ValueError("--since is empty")
    82 +    match = _DURATION.match(text)
    83 +    if match:
    84 +        amount = int(match.group(1))
    85 +        return now - timedelta(seconds=amount * _UNIT_SECONDS[match.group(2).lower()])
    86 +    try:
    87 +        parsed = datetime.fromisoformat(text.replace("Z", "+00:00"))
    88 +    except ValueError as exc:
    89 +        raise ValueError(
    90 +            "--since %r is neither a duration (6h, 90m, 2d, 3w) nor an ISO "
    91 +            "datetime" % value
    92 +        ) from exc
    93 +    return parsed if parsed.tzinfo else parsed.replace(tzinfo=timezone.utc)
    94 +
    95 +
    96 +def parse_timestamp(value: str) -> datetime | None:
    97 +    """A forge timestamp, or None when the field is absent or unreadable.
    98 +
    99 +    None is a real answer here rather than an error: a pull request with no
   100 +    `mergedAt` is simply not merged, and the caller must not mistake that for a
   101 +    broken snapshot.
   102 +    """
   103 +    if not isinstance(value, str) or not value.strip():
   104 +        return None
   105 +    try:
   106 +        parsed = datetime.fromisoformat(value.strip().replace("Z", "+00:00"))
   107 +    except ValueError:
   108 +        return None
   109 +    return parsed if parsed.tzinfo else parsed.replace(tzinfo=timezone.utc)
   110 +

   112 +# --- Parsers ----------------------------------------------------------------
   113 +# Every parser below takes TEXT, not a subprocess. That is what lets the suite
   114 +# cover them with recorded output and no network.
   115 +
   116 +
   117 +def parse_ls_remote(text: str) -> list[str]:
   118 +    """Branch names from `git ls-remote --heads`, sorted.
   119 +
   120 +    The remote is the only coordination point between machines, so this is the
   121 +    authoritative branch list. `git branch -r` is a local cache and is NOT used.
   122 +    """
   123 +    names = []
   124 +    for line in (text or "").splitlines():
   125 +        match = _LS_REMOTE_LINE.match(line.strip())
   126 +        if match:
   127 +            names.append(match.group(2))
   128 +    return sorted(set(names))
   129 +
   130 +
   131 +def parse_porcelain(text: str) -> list[dict]:
   132 +    """Uncommitted paths from `git status --porcelain`.
   133 +
   134 +    A rename prints both halves; the NEW path is reported, because that is the
   135 +    file now on disk. The two status characters are kept verbatim so a caller
   136 +    can tell a staged change from an unstaged one.
   137 +    """
   138 +    out = []
   139 +    for line in (text or "").splitlines():
   140 +        if len(line) < 4:
   141 +            continue
   142 +        status, path = line[:2], line[3:].strip()
   143 +        if _RENAME_ARROW in path:
   144 +            path = path.split(_RENAME_ARROW, 1)[1]
   145 +        out.append({"status": status.strip() or status, "path": path.strip('"')})
   146 +    return out
   147 +
   148 +
   149 +def parse_pr_list(text: str) -> list[dict]:
   150 +    """Pull requests from `gh pr list --json ...`.
   151 +
   152 +    Only the fields a claim can be checked against are kept. A malformed entry
   153 +    is dropped rather than guessed at: a half-read record is worse than a
   154 +    missing one, because it looks like corroboration.
   155 +    """
   156 +    try:
   157 +        rows = json.loads(text or "[]")
   158 +    except json.JSONDecodeError as exc:
   159 +        raise RepoError("gh returned output that is not JSON: %s" % exc) from exc
   160 +    if not isinstance(rows, list):
   161 +        raise RepoError("gh returned %s, expected a list" % type(rows).__name__)
   162 +    out = []
   163 +    for row in rows:
   164 +        if not isinstance(row, dict) or not isinstance(row.get("number"), int):
   165 +            continue
   166 +        out.append({
   167 +            "number": row["number"],
   168 +            "title": str(row.get("title") or ""),
   169 +            "head_branch": str(row.get("headRefName") or ""),
   170 +            "created_at": str(row.get("createdAt") or ""),
   171 +            "merged_at": str(row.get("mergedAt") or ""),
   172 +            "url": str(row.get("url") or ""),
   173 +        })
   174 +    return sorted(out, key=lambda r: r["number"])
   175 +
   176 +
   177 +def filter_merged_since(rows: list[dict], since: datetime) -> list[dict]:
   178 +    """Merged pull requests inside the window.
   179 +
   180 +    An unparseable or absent `merged_at` drops the row. A pull request that
   181 +    cannot be dated cannot corroborate a claim about when it merged.
   182 +    """
   183 +    out = []
   184 +    for row in rows:
   185 +        merged = parse_timestamp(row.get("merged_at", ""))
   186 +        if merged >= since:
   187 +            out.append(row)
   188 +    return out
   189 +

   191 +# --- Formatting -------------------------------------------------------------
   192 +
   193 +
   194 +def _truncate(text: str, width: int) -> str:
   195 +    """Collapse whitespace and cut to `width`.
   196 +
   197 +    The marker is ASCII on purpose. A Windows console defaults to a code page
   198 +    that cannot encode a typographic ellipsis, and the table is written to
   199 +    stderr where an encoding error would replace the diagnostic with a
   200 +    traceback.
   201 +    """
   202 +    text = " ".join(str(text).split())
   203 +    return text if len(text) <= width else text[: width - 3] + "..."
   204 +
   205 +
   206 +def format_table(snapshot: dict) -> str:
   207 +    """The compact human view. Goes to stderr; stdout carries the JSON."""
   208 +    lines = []
   209 +    for repo in snapshot.get("repos", []):
   210 +        lines.append("== %s" % repo.get("path", "?"))
   211 +        if repo.get("error"):
   212 +            lines.append("   ERROR: %s" % repo["error"])
   213 +            continue
   214 +        lines.append("   branch      %s" % (repo.get("current_branch") or "?"))
   215 +        head = repo.get("last_commit") or {}
   216 +        lines.append("   last commit %s %s" % (
   217 +            head.get("sha", "?")[:8], _truncate(head.get("subject", ""), 56)))
   218 +        dirty = repo.get("uncommitted") or []
   219 +        if dirty:
   220 +            lines.append("   uncommitted %d path(s)" % len(dirty))
   221 +            for entry in dirty[:10]:
   222 +                lines.append("     %-2s %s" % (entry["status"], entry["path"]))
   223 +            if len(dirty) > 10:
   224 +                lines.append("     ... and %d more" % (len(dirty) - 10))
   225 +        else:
   226 +            lines.append("   uncommitted none")
   227 +        remote = repo.get("remote_branches") or []
   228 +        lines.append("   remote      %d branch(es): %s" % (
   229 +            len(remote), _truncate(", ".join(remote), 60) or "none"))
   230 +        for label, key in (("open PR", "open_prs"), ("merged PR", "merged_prs")):
   231 +            rows = repo.get(key) or []
   232 +            if not rows:
   233 +                lines.append("   %-11s none" % (label + "s"))
   234 +                continue
   235 +            lines.append("   %-11s %d" % (label + "s", len(rows)))
   236 +            for row in rows:
   237 +                lines.append("     #%-5d %-28s %s" % (
   238 +                    row["number"], _truncate(row["head_branch"], 28),
   239 +                    _truncate(row["title"], 44)))
   240 +    lines.append("")
   241 +    lines.append("A claim with no matching record above is a fabrication or a "
   242 +                 "silent failure.")
   243 +    return "\n".join(lines)

   246 +# --- Collection -------------------------------------------------------------
   247 +
   248 +
   249 +def _run(args: list[str], cwd: str) -> str:
   250 +    """Run a read-only command and return its stdout. Raises RepoError."""
   251 +    try:
   252 +        proc = subprocess.run(
   253 +            args, cwd=cwd, capture_output=True, text=True, timeout=_TIMEOUT
   254 +        )
   255 +    except FileNotFoundError as exc:
   256 +        raise RepoError("%s is not on PATH" % args[0]) from exc
   257 +    except subprocess.TimeoutExpired as exc:
   258 +        raise RepoError("%s did not finish within %ss" % (args[0], _TIMEOUT)) from exc
   259 +    if proc.returncode != 0:
   260 +        raise RepoError("%s exited %d: %s" % (
   261 +            " ".join(args[:3]), proc.returncode, (proc.stderr or "").strip()[:300]))
   262 +    return proc.stdout

   265 +_PR_FIELDS = "number,title,headRefName,createdAt,mergedAt,url"
   266 +
   267 +
   268 +def collect(path: str, since: datetime, runner=_run) -> dict:
   269 +    """One repo's ground-truth snapshot. `runner` is injected for the suite."""
   270 +    repo: dict = {"path": path}
   271 +    try:
   272 +        repo["current_branch"] = runner(
   273 +            ["git", "-C", path, "rev-parse", "--abbrev-ref", "HEAD"], path).strip()
   274 +        head = runner(
   275 +            ["git", "-C", path, "log", "-1", "--format=%H%x1f%s%x1f%cI"], path).strip()
   276 +        sha, _, rest = head.partition("\x1f")
   277 +        subject, _, committed = rest.partition("\x1f")
   278 +        repo["last_commit"] = {"sha": sha, "subject": subject,
   279 +                               "committed_at": committed}
   280 +        repo["uncommitted"] = parse_porcelain(
   281 +            runner(["git", "-C", path, "status", "--porcelain"], path))
   282 +        repo["remote_branches"] = parse_ls_remote(
   283 +            runner(["git", "-C", path, "ls-remote", "--heads", "origin"], path))
   284 +        open_prs = parse_pr_list(runner(
   285 +            ["gh", "pr", "list", "--state", "open", "--limit", "100",
   286 +             "--json", _PR_FIELDS], path))
   287 +        merged = parse_pr_list(runner(
   288 +            ["gh", "pr", "list", "--state", "merged", "--limit", "100",
   289 +             "--json", _PR_FIELDS], path))
   290 +        repo["open_prs"] = open_prs
   291 +        repo["merged_prs"] = filter_merged_since(merged, since)
   292 +    except RepoError as exc:
   293 +        repo["error"] = str(exc)
   294 +    return repo
   295 +
   296 +
   297 +def build_snapshot(paths: list[str], since: datetime, runner=_run) -> dict:
   298 +    return {
   299 +        "generated_at": datetime.now(timezone.utc).isoformat(),
   300 +        "since": since.isoformat(),
   301 +        "repos": [collect(p, since, runner) for p in paths],
   302 +    }
